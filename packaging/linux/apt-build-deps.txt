# System build + bundling dependencies for the arraw Linux AppImage (Ubuntu 24.04).
#
# Single source of truth, consumed by both AppImage build paths so the list never
# drifts between them:
#   - .github/workflows/release.yml   (the shipped release)
#   - packaging/linux/Containerfile   (local `just appimage` repro)
#
# One package per line; blank lines and `#` comments are ignored. The Containerfile
# additionally installs python3-pip for its in-image aqt Qt provisioning; the CI job
# gets Qt from install-qt-action instead, so that one stays out of this shared list.

# Toolchain + packaging utilities
build-essential
cmake
ninja-build
file
wget
ca-certificates
appstream
libdbus-1-3

# RAW decode, color management, lens corrections (lensfun lib + DB, #53), and
# exported metadata embedding.
libraw-dev
liblcms2-dev
liblensfun-dev
liblensfun-data-v1
libexiv2-dev

# OpenGL / Vulkan / xkb / fontconfig that official Qt's imported targets reference
